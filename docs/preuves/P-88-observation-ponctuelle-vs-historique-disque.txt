# 3) Tendance calculée par Prometheus (Mio/jour consommés, sur les 6 dernières heures ; projection à 30 jours)
$ query= deriv(node_filesystem_avail_bytes{mountpoint="/", host=~".+"}[6h]) * -86400 / 2^20
